	function cancelarTramite() {
        if (confirm('Si cancela su solicitud, esta no se registrar\u00E1. Desea continuar?')) {
            document.forma.accion.value = 'cancelar';
            document.forma.submit();
        }

function validarNum(b, obj, valor, a){		
	if ((obj=="")||(obj==0)){
		alert ("Debe ingresar un valor v\u00E1lido.");
	}else{
		var bNumeric=true;
		for (var ii = 0 ; ii < obj.length ; ii++) {
			if(0 > ('0123456789').indexOf(obj.substring(ii, ii+1))) {
				bNumeric=false;
				break;
			}
		}
		if (bNumeric!=true){
			alert ("Debe ingresar un n\u00FAmero.");
		}else{
			if (b==1){
				if (obj > valor){
					alert ("No puede ingresar un n\u00FAmero superior al n\u00FAmero total de registros.");
				}else{	
					document.forma.accion.value='cambiar-lineas-pagina';
					document.forma.submit();
				}
			}else{
				if (obj > a){
					alert ("No puede ingresar un n\u00FAmero superior al n\u00FAmero total de p\u00E1ginas.");
				}else{	
					document.forma.accion.value='ir-a-pagina';
					document.forma.submit();
				}
			}
		}
	}
}

	function getReadyStateHandler(xmlHttpRequest) {

  function isNumeric(valor) {
    var bNumeric=true;

      if(0 > ('0123456789').indexOf(valor.substring(ii, ii+1))) {

        break;
